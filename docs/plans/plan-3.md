```js
const REGEX_AFTER_WORD = new Set(['return', 'typeof', 'case', 'do', 'else', 'in', 'of', 'new', 'delete', 'void', 'throw', 'instanceof', 'yield', 'await']);
export function branchPointWeight(src, start, end) {
  let n = 0;
  const braces = [];
  const isId = (c) => /[\w$]/.test(c);
  let i = start;
  let prev = ''; // last significant token: a word, or a punctuator character; ')' also stands for any value token
  const regexAllowed = () => (prev === '' ? true : /^[\w$]+$/.test(prev) ? REGEX_AFTER_WORD.has(prev) : !(prev === ')' || prev === ']'));
  const template = () => {
    while (i < end) {
      if (src[i] === '\\') i += 2;
      else if (src[i] === '`') { i++; return; }
      else if (src[i] === '$' && src[i + 1] === '{') { braces.push(true); i += 2; return; }
      else i++;
    }
  };
  const regex = () => { // i at the opening '/'
    i++;
    let cls = false;
    while (i < end && src[i] !== '\n') {
      const ch = src[i];
      if (ch === '\\') { i += 2; continue; }
      if (cls) { if (ch === ']') cls = false; } else if (ch === '[') cls = true; else if (ch === '/') { i++; break; }
      i++;
    }
    while (i < end && /[A-Za-z]/.test(src[i])) i++;
  };
  while (i < end) {
    const c = src[i];
    const d = src[i + 1];
    if (c === '/' && d === '/') { while (i < end && src[i] !== '\n') i++; }
    else if (c === '/' && d === '*') { const j = src.indexOf('*/', i + 2); i = j < 0 || j + 2 > end ? end : j + 2; }
    else if (c === '/' && regexAllowed()) { regex(); prev = ')'; }
    else if (c === "'" || c === '"') { i++; while (i < end && src[i] !== c && src[i] !== '\n') i += src[i] === '\\' ? 2 : 1; i++; prev = ')'; }
    else if (c === '`') { i++; template(); prev = ')'; }
    else if (c === '{') { braces.push(false); i++; prev = '{'; }
    else if (c === '}') { i++; if (braces.pop()) { template(); prev = ')'; } else prev = '}'; }
    else if ((c === '?' && d === '?') || (c === '|' && d === '|') || (c === '&' && d === '&')) { n++; i += 2; prev = c; }
    else if (c === '?') { if (d !== '.' || /\d/.test(src[i + 2] ?? '')) n++; i++; prev = '?'; }
    else if (isId(c) && !isId(src[i - 1] ?? '')) {
      let j = i;
      while (j < end && isId(src[j])) j++;
      const word = src.slice(i, j);
      if (BRANCH_WORDS.has(word) && !isId(src[j] ?? '')) n++;
      prev = word;
      i = j;
    } else { if (!/\s/.test(c)) prev = c; i++; }
  }
  return n;
}
```

A closing `}` of a `${…}` resumes the template, which is a value (`prev = ')'`). A block `}` allows a regex. Keep the existing doc comment above the function and add one line to it: "Regex literal bodies are skipped. After a value token (identifier, number, `)`, `]`, string, template) a `/` is division."

```js
// Replay observation boundary (Codex P2-12): what changed anywhere in the sandbox, and stable diagnostics.
import { readdirSync, readFileSync, lstatSync, readlinkSync, existsSync, statSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { join } from 'node:path';

const sha = (b) => createHash('sha256').update(b).digest('hex').slice(0, 16);
const isGitDir = (p) => { try { return statSync(join(p, 'HEAD')).isFile() && statSync(join(p, 'objects')).isDirectory(); } catch { return false; } };

export function snapshotTree(root, { skip }) {
  const out = new Map();
  const walk = (abs, rel) => {
    let ents;
    try { ents = readdirSync(abs, { withFileTypes: true }); } catch { if (rel) out.set(`${rel}/`, 'u'); return; }
    for (const e of ents.sort((a, b) => (a.name < b.name ? -1 : a.name > b.name ? 1 : 0))) {
      const r = rel ? `${rel}/${e.name}` : e.name;
      if (skip(r)) continue;
      const p = join(abs, e.name);
      const st = lstatSync(p);
      if (st.isSymbolicLink()) out.set(r, `l:${readlinkSync(p)}`);
      else if (st.isDirectory()) {
        if (isGitDir(p)) { const inner = snapshotTree(p, { skip: () => false }); out.set(`${r}/`, `g:${sha([...inner].map((x) => x.join('=')).join('\n'))}`); }
        else walk(p, r);
      } else if (st.isFile()) { try { out.set(r, `f:${sha(readFileSync(p))}`); } catch { out.set(r, 'u'); } }
    }
  };
  if (existsSync(root)) walk(root, '');
  return out;
}

export function treeChanges(before, after) {
  const lines = [];
  for (const [k, v] of after) { if (!before.has(k)) lines.push(['A', k]); else if (before.get(k) !== v) lines.push(['M', k]); }
  for (const k of before.keys()) if (!after.has(k)) lines.push(['D', k]);
  return lines.sort((a, b) => (a[1] < b[1] ? -1 : a[1] > b[1] ? 1 : 0)).map(([t, k]) => `${t} ${k}`);
}

export function normalizeDiagnostics(text) {
  const lines = text.split('\n');
  const out = [];
  for (let i = 0; i < lines.length; i++) {
    const l = lines[i];
    if (/^\(node:\d+\) \w*Warning: /.test(l)) { if (/^\(Use `node --trace-/.test(lines[i + 1] ?? '')) i++; continue; }
    if (/^(file:\/\/|\/|<IMPL>|[A-Za-z]:\\).*:\d+$/.test(l) && (lines[i + 2] ?? '').trim() === '^'.repeat((lines[i + 2] ?? '').trim().length) && (lines[i + 2] ?? '').trim()) { out.push('<uncaught-at>'); i += 2; continue; }
    out.push(/^\s+at .*$/.test(l) ? '    at <frame>' : l);
  }
  return out.join('\n');
}
```

```js
// B2 behaviour migrations (spec 14.8). Each lands alone behind its id. TOKENOMICS_MIGRATIONS selects
// them: unset = every landed migration; `none` = none (the B1 extraction — also the B4 rescue
// release's writer, spec 18.5); `a,b` = exactly those. It is the harness/rescue lever; production
// per-migration switches (spec 18.5) are settings work in B3/B4. STDLIB ONLY.
export const MIGRATIONS = Object.freeze([
  ['exact-ids', 1], ['shared-allocation', 2], ['copilot-inclusive', 3], ['null-unobservable', 4],
  ['anon-complete', 5], ['fold-flags', 6], ['active-minutes', 7], ['report-windows', 8],
  ['scope-resolution', 9], ['reports-location', 10], ['writer', 11],
].map(([id, item]) => Object.freeze({ id, item, landed: false })));

export function enabledMigrations(env = process.env) {
  const raw = env.TOKENOMICS_MIGRATIONS;
  const landed = MIGRATIONS.filter((m) => m.landed).map((m) => m.id);
  if (raw === undefined || raw.trim() === '') return new Set(landed);
  if (raw.trim() === 'none') return new Set();
  const ids = raw.split(',').map((s) => s.trim()).filter(Boolean);
  for (const id of ids) {
    const m = MIGRATIONS.find((x) => x.id === id);
    if (!m) throw new Error(`tokenomics: unknown migration '${id}' in TOKENOMICS_MIGRATIONS`);
    if (!m.landed) throw new Error(`tokenomics: migration '${id}' has not landed`);
  }
  return new Set(ids);
}

export const migrationOn = (id, env = process.env) => enabledMigrations(env).has(id);
```
